      function init() {
        try {
          if (!ctx) {
            const AC = window.AudioContext || window.webkitAudioContext;
            if (!AC) return false;
            ctx = new AC();

            masterGain = ctx.createGain();
            bgmGain = ctx.createGain();
            sfxGain = ctx.createGain();

            masterGain.gain.setValueAtTime(masterVol, ctx.currentTime);
            bgmGain.gain.setValueAtTime(bgmVol, ctx.currentTime);
            sfxGain.gain.setValueAtTime(1, ctx.currentTime);

            bgmGain.connect(masterGain);
            sfxGain.connect(masterGain);
            masterGain.connect(ctx.destination);

            const len = ctx.sampleRate * 3;
            noiseBuf = ctx.createBuffer(1, len, ctx.sampleRate);
            const data = noiseBuf.getChannelData(0);
            let lastOut = 0;
            for (let i = 0; i < len; i++) {
              const white = Math.random() * 2 - 1;
              lastOut = (lastOut + (0.04 * white)) / 1.04;
              data[i] = white * 0.7 + lastOut * 0.6;
            }
          }
          if (ctx.state === 'suspended') ctx.resume();
          return true;
        } catch {
          return false;
        }
      }

      function syncGains() {
        if (!ctx) return;
        const now = ctx.currentTime;
        if (masterGain) masterGain.gain.setTargetAtTime(masterVol, now, 0.03);
        if (bgmGain) bgmGain.gain.setTargetAtTime(bgmVol, now, 0.03);
      }

      function startAmbience() {
        if (!init() || !noiseBuf || ambSrc) return;
        try {
          ambSrc = ctx.createBufferSource();
          ambSrc.buffer = noiseBuf;
          ambSrc.loop = true;

          ambFilter = ctx.createBiquadFilter();
          ambFilter.type = 'lowpass';
          ambFilter.frequency.setValueAtTime(320, ctx.currentTime);

          ambGain = ctx.createGain();
          ambGain.gain.setValueAtTime(0.018, ctx.currentTime);

          ambSrc.connect(ambFilter).connect(ambGain).connect(sfxGain);
          ambSrc.start();
          updateAmbienceTone();
        } catch {}
      }

      function stopAmbience() {
        try {
          if (ambSrc) { ambSrc.stop(); ambSrc.disconnect(); ambSrc = null; }
        } catch { ambSrc = null; }
      }

      function tone(freq, dur, { type = 'sine', gain = 0.16, endFreq = null, delay = 0, pan = 0, bus = 'sfx' } = {}) {
        if (masterVol <= 0 || !init()) return;
        if (bus === 'bgm' && bgmVol <= 0) return;
        const now = ctx.currentTime + delay;
        const osc = ctx.createOscillator();
        const g = ctx.createGain();
        osc.type = type;
        osc.frequency.setValueAtTime(freq, now);
        if (endFreq) osc.frequency.exponentialRampToValueAtTime(Math.max(20, endFreq), now + dur);

        g.gain.setValueAtTime(0.0001, now);
        g.gain.linearRampToValueAtTime(gain, now + Math.min(0.015, dur * 0.25));
        g.gain.exponentialRampToValueAtTime(0.0001, now + dur);

        const targetBus = bus === 'bgm' ? bgmGain : sfxGain;
        if (ctx.createStereoPanner) {
          const panner = ctx.createStereoPanner();
          panner.pan.setValueAtTime(Math.max(-1, Math.min(1, pan)), now);
          osc.connect(g).connect(panner).connect(targetBus);
        } else {
          osc.connect(g).connect(targetBus);
        }
        osc.start(now);
        osc.stop(now + dur + 0.02);
      }

        const g = ctx.createGain();
        g.gain.setValueAtTime(0.0001, now);
        g.gain.linearRampToValueAtTime(gain, now + Math.min(0.01, dur * 0.2));
        g.gain.exponentialRampToValueAtTime(0.0001, now + dur);

        if (ctx.createStereoPanner) {
          const panner = ctx.createStereoPanner();
          panner.pan.setValueAtTime(Math.max(-1, Math.min(1, pan)), now);
          src.connect(filter).connect(g).connect(panner).connect(sfxGain);
        } else {
          src.connect(filter).connect(g).connect(sfxGain);
        }
        src.start(now);
        src.stop(now + dur + 0.02);
      }

    function focusGameScreen() {
      if (modalOpen) return;
      const gs = document.getElementById('gameScreen');
      if (gs && !gs.classList.contains('hidden')) {
        gs.focus({ preventScroll: true });
      }

    function announce(text) {
      const el = document.getElementById('srAnnouncer');
      clearTimeout(announceTimer);
      announceToggle = !announceToggle;
      const clean = String(text || '').trim();
      el.textContent = '';
      announceTimer = setTimeout(() => {
        el.textContent = announceToggle ? clean : clean + '\u00A0';
      }, 25);
    }

    function addLog(msg, cls = '', speak = false) {
      logs.push({ msg, cls });
      if (logs.length > 60) logs.shift();
      const box = document.getElementById('gameLog');
      if (box) {
        box.innerHTML = logs.map(l => `<div class="${l.cls}">${l.msg}</div>`).join('');
        box.scrollTop = box.scrollHeight;
      }
      if (speak) announce(stripHtml(msg));
    }

    function stripHtml(html) {
      const tmp = document.createElement('div');
      tmp.innerHTML = html;
      return tmp.textContent || tmp.innerText || '';
    }

    function showInspect(title, html, speakText = null) {
      document.getElementById('inspectTitle').textContent = title;
      document.getElementById('inspectBody').innerHTML = html;
      announce(speakText || stripHtml(html));
    }

    // ==================== AUTH & SAVE ====================
    function getAccounts() {
      try { return JSON.parse(localStorage.getItem(STORAGE_ACCOUNTS)) || {}; }
      catch { return {}; }
    }

    function saveData() {
      if (!currentUser || !player) return;
      const acc = getAccounts();
      if (acc[currentUser]) {
        acc[currentUser].data = player;
        localStorage.setItem(STORAGE_ACCOUNTS, JSON.stringify(acc));
      }
    }

    function manualSave() {
      saveData();
      SoundFX.play('save');
      addLog('💾 บันทึกความคืบหน้าเรียบร้อยแล้ว', 'c-win', true);
    }

    function setAuthMode(mode) {
      authMode = mode;
      document.getElementById('tabLogin').classList.toggle('active', mode === 'login');
      document.getElementById('tabReg').classList.toggle('active', mode === 'register');
      document.getElementById('authBtn').textContent = mode === 'login' ? 'เข้าสู่ยุทธภพ' : 'สร้างตัวละครใหม่';
      document.getElementById('authMsg').textContent = '';
    }

    function submitAuth(e) {
      e.preventDefault();
      SoundFX.init();
      const name = document.getElementById('inName').value.trim();
      const pass = document.getElementById('inPass').value;
      const msg = document.getElementById('authMsg');
      if (name.length < 2) { msg.textContent = 'ชื่อตัวละครต้องมีอย่างน้อย 2 ตัวอักษร'; return; }

      const accounts = getAccounts();
      const key = name.toLowerCase();
      const encoded = btoa(unescape(encodeURIComponent(pass)));

      if (authMode === 'register') {
        if (accounts[key]) { msg.textContent = 'ชื่อนี้มีผู้ใช้งานแล้ว กรุณาใช้ชื่ออื่น'; return; }
        const p = createFreshPlayer(name);
        accounts[key] = { pass: encoded, data: p };
        localStorage.setItem(STORAGE_ACCOUNTS, JSON.stringify(accounts));
        startSession(key, p);
      } else {
        if (!accounts[key] || accounts[key].pass !== encoded) {
          SoundFX.play('bump');
          msg.textContent = 'ชื่อตัวละครหรือรหัสผ่านไม่ถูกต้อง';
          return;
        }
        startSession(key, accounts[key].data);
      }
    }

    // ==================== STAT HELPERS ====================
    function getRealm(lv) {
      let res = REALMS[0];
      for (const r of REALMS) if (lv >= r.minLv) res = r;
      return res;
    }

    function maxHp() { return player.maxHp + (player.foundationBonus?.hp || 0); }
    function maxSp() { return player.maxSpirit + (player.foundationBonus?.sp || 0); }
    function totalAtk() {
      const wAtk = player.equippedWeapon && ITEMS[player.equippedWeapon] ? ITEMS[player.equippedWeapon].atk : 0;
      return player.baseAtk + wAtk + (player.foundationBonus?.atk || 0);
    }

    function addItem(id, qty = 1) { player.inventory[id] = (player.inventory[id] || 0) + qty; }
    function removeItem(id, qty = 1) {
      if (!player.inventory[id]) return false;
      player.inventory[id] -= qty;
      if (player.inventory[id] <= 0) delete player.inventory[id];
      return true;
    }

    function gainExp(amt) {
      player.exp += amt;
      let up = false;
      let extraMsg = '';
      const oldRealm = getRealm(player.level).name;
      while (player.exp >= player.maxExp) {
        player.exp -= player.maxExp;
        player.level += 1;
        player.maxExp = Math.floor(player.maxExp * 1.28 + 30);
        player.maxHp += 26;
        player.maxSpirit += 10;
        player.baseAtk += 4;
        player.hp = maxHp();
        player.spirit = maxSp();
        up = true;
        const newSkill = SKILLS.find(s => s.unlockLv === player.level);
        if (newSkill) {
          extraMsg += ` ปลดล็อกสกิลใหม่ ช่อง ${newSkill.key} ${newSkill.name}`;
          addLog(`📜 ปลดล็อกสกิลใหม่ ช่อง [${newSkill.key}]: ${newSkill.name}!`, 'c-info');
        }
      }
      if (up) {
        SoundFX.play('levelup');
        const newRealm = getRealm(player.level).name;
        const msg = newRealm !== oldRealm
          ? `⚡ ทะลวงขอบเขตพลังจิตใหม่! บรรลุ ${newRealm} (เลเวล ${player.level})${extraMsg}`
          : `✨ อัปเลเวล ${player.level}! ฟื้นฟูพลังเต็มเปี่ยม${extraMsg}`;
        addLog(msg, 'c-win');
        return msg;
      }
      return '';
    }

    // ==================== MAP & MOVEMENT ====================
    function initMapMonsters() {
      ['zone_1', 'zone_2', 'zone_3', 'zone_4'].forEach(zId => {
        activeMonsters[zId] = {};
        MAPS[zId].monsterSpots.forEach(([x, y, idx]) => {
          activeMonsters[zId][`${x},${y}`] = idx;
        });
      });
    }

    function respawnMonstersInZone(zId) {
      if (!MAPS[zId]?.monsterSpots) return;
      activeMonsters[zId] = activeMonsters[zId] || {};
      MAPS[zId].monsterSpots.forEach(([x, y, idx]) => {
        if (activeMonsters[zId][`${x},${y}`] === undefined) {
          activeMonsters[zId][`${x},${y}`] = idx;
        }
      });
    }

    function getTileInfo(mapId, x, y) {
      const m = MAPS[mapId];
      const key = `${x},${y}`;
      if (m.tiles[key]) return m.tiles[key];
      if (m.type === 'wild' && activeMonsters[mapId] && activeMonsters[mapId][key] !== undefined) {
        const pool = MONSTER_POOLS[mapId];
        const mon = pool[activeMonsters[mapId][key]];
        return { icon: mon.emoji, label: mon.name, kind: 'monster', monIndex: activeMonsters[mapId][key], desc: `พบอสูร ${mon.name} (Lv.${mon.lv})` };
      }
      return {
        icon: m.type === 'city' ? '·' : '🌿',
        label: m.type === 'city' ? 'ถนนในเมือง' : 'ทุ่งอสูร',
        kind: 'empty',
        desc: m.type === 'city' ? 'ถนนว่างในเมือง' : 'พื้นที่โล่งนอกเมือง'
      };
    }

    function dirName(dx, dy) {
      if (dy < 0) return 'ทิศเหนือ';
      if (dy > 0) return 'ทิศใต้';
      if (dx < 0) return 'ทิศตะวันตก';
      return 'ทิศตะวันออก';
    }

    function movePlayer(dx, dy) {
      if (combat || modalOpen) return;
      const m = MAPS[player.mapId];
      const nx = player.x + dx;
      const ny = player.y + dy;
      const pan = dx * 0.45;

      if (nx < 0 || nx >= m.size || ny < 0 || ny >= m.size) {
        SoundFX.play('bump', pan);
        const wallMsg = `ชนขอบแผนที่${dirName(dx, dy)} พิกัดปัจจุบัน (${player.x}, ${player.y})`;
        document.getElementById('uiTileDesc').textContent = wallMsg;
        announce(wallMsg);
        return;
      }

      player.x = nx;
      player.y = ny;
      player.steps = (player.steps || 0) + 1;

      if (m.type === 'wild' && player.steps % 4 === 0) {
        respawnMonstersInZone(player.mapId);
      }

      const info = getTileInfo(player.mapId, nx, ny);
      saveData();
      renderMap();

      SoundFX.play(m.type === 'city' ? 'step_stone' : 'step_grass', pan);

      if (info.kind === 'monster') {
        startCombat(info.monIndex, `${nx},${ny}`, `เดิน${dirName(dx, dy)} พิกัด (${nx}, ${ny})`);
        return;
      }

      if (m.type === 'wild' && info.kind === 'empty' && Math.random() < 0.14) {
        const pool = MONSTER_POOLS[player.mapId];
        const rIdx = Math.floor(Math.random() * pool.length);
        startCombat(rIdx, null, `เดิน${dirName(dx, dy)} พิกัด (${nx}, ${ny}) ถูกซุ่มโจมตี!`);
        return;
      }

      let stepSpeech = `เดิน${dirName(dx, dy)} พิกัด (${nx}, ${ny}) ${info.label}`;
      if (info.kind !== 'empty' && info.kind !== 'landmark') {
        SoundFX.play('poi');
        stepSpeech += ' กด Enter เพื่อเข้าใช้งาน';
      }
      document.getElementById('uiTileDesc').textContent = `${info.label}: ${info.desc}`;
      announce(stepSpeech);
    }

    function interactCurrentTile() {
      if (combat || modalOpen) return;
      const info = getTileInfo(player.mapId, player.x, player.y);

      if (info.kind === 'empty' || info.kind === 'landmark') {
        SoundFX.play('bump');
        const msg = `พิกัด (${player.x}, ${player.y}) เป็น ${info.label} ไม่มีร้านค้าหรือประตูที่นี่`;
        addLog(msg, 'c-info', true);
        return;
      }

      if (info.kind === 'gate') {
        if (player.level < (info.minLv || 1)) {
          SoundFX.play('bump');
          addLog(`🔒 ต้องการเลเวล ${info.minLv} เพื่อผ่าน${info.label} (ปัจจุบันคุณ Lv.${player.level})`, 'c-bad', true);
          return;
        }
        SoundFX.play('gate');
        player.mapId = info.toMap;
        player.x = info.toX;
        player.y = info.toY;
        respawnMonstersInZone(player.mapId);
        SoundFX.updateAmbienceTone();
        saveData();
        renderAll();
        focusGameScreen();
        addLog(`🚪 เดินทางมาถึง ${MAPS[player.mapId].name} พิกัด (${player.x}, ${player.y})`, 'c-win', true);
        return;
      }

      if (info.kind === 'inn') {
        SoundFX.play('heal');
        if (MAPS[player.mapId].type === 'city') {
          player.hp = maxHp();
          player.spirit = maxSp();
          addLog(`🏠 พักผ่อนที่ ${info.label} ฟื้นฟู HP และพลังจิตเต็ม 100% แล้ว!`, 'c-win', true);
        } else {
          const rHp = Math.round(maxHp() * 0.35);
          const rSp = Math.round(maxSp() * 0.35);
          player.hp = Math.min(maxHp(), player.hp + rHp);
          player.spirit = Math.min(maxSp(), player.spirit + rSp);
          addLog(`⛺ พักหายใจที่ ${info.label} ฟื้น HP +${rHp}, พลังจิต +${rSp}`, 'c-win', true);
        }
        saveData();
        renderHeader();
        return;
      }

      if (info.kind === 'foundation') {
        openFoundationModal();
        return;
      }

      if (info.kind.startsWith('shop_')) {
        openShopModal(info);
      }
    }

    function searchNearbyMonster() {

    // ==================== COMBAT (1-0 SKILLS) ====================
    function startCombat(monIndex, spotKey, prefix = '') {
      const template = MONSTER_POOLS[player.mapId][monIndex];
      combat = {
        spotKey,
        stunned: false,
        guarding: false,
        enemy: { ...template, maxHp: template.hp }
      };
      SoundFX.play('encounter');
      SoundFX.updateAmbienceTone();
      renderAll();
      focusGameScreen();
      addLog(`⚔️ ${prefix ? prefix + ' ' : ''}เผชิญหน้ากับ ${template.name} (Lv.${template.lv}, HP ${template.hp}, โจมตี ${template.atk}) กดเลข 1 ถึง 0 ใช้สกิล`, 'c-bad', true);
    }

    function useSkillByKey(key) {
      if (!combat || modalOpen) return;
      const skill = SKILLS.find(s => s.key === key);
      if (!skill) return;

      if (player.level < skill.unlockLv) {
        SoundFX.play('bump');
        announce(`สกิลช่อง ${key} ${skill.name} ยังไม่ปลดล็อก ต้องใช้เลเวล ${skill.unlockLv}`);
        return;
      }
      if (player.spirit < skill.cost) {
        SoundFX.play('bump');
        announce(`พลังจิตไม่พอสำหรับ ${skill.name} ต้องการ ${skill.cost} แต่มี ${player.spirit}`);
        return;
      }

      player.spirit -= skill.cost;
      combat.guarding = false;
      const e = combat.enemy;
      let report = '';

      if (key === '1') {
        SoundFX.play('slash');
        const dmg = Math.round(totalAtk() * (0.95 + Math.random() * 0.18));
        e.hp -= dmg;
        report = `ใช้ ${skill.name} โจมตี ${e.name} เสียหาย ${dmg}`;
      } else if (key === '2') {
        SoundFX.play('shield');
        combat.guarding = 0.55;
        const h = Math.round(maxHp() * 0.12);
        const s = Math.round(maxSp() * 0.20);
        player.hp = Math.min(maxHp(), player.hp + h);
        player.spirit = Math.min(maxSp(), player.spirit + s);
        report = `ใช้ ${skill.name} ฟื้น HP +${h}, จิต +${s} และลดดาเมจ 45%`;
      } else if (key === '3') {
        SoundFX.play('spirit_wave');
        const dmg = Math.round(totalAtk() * 1.85 + maxSp() * 0.25);
        e.hp -= dmg;
        report = `ใช้ ${skill.name} สร้างความเสียหาย ${dmg}`;
      } else if (key === '4') {
        SoundFX.play('crit_slash');
        const crit = Math.random() < 0.5;
        const dmg = Math.round(totalAtk() * 2.2 * (crit ? 1.55 : 1));
        e.hp -= dmg;
        report = `ใช้ ${skill.name}${crit ? ' ติดคริติคอล' : ''} สร้างความเสียหาย ${dmg}`;
      } else if (key === '5') {
        SoundFX.play('spirit_wave');
        const dmg = Math.round(totalAtk() * 1.9);
        const heal = Math.round(dmg * 0.35);
        e.hp -= dmg;
        player.hp = Math.min(maxHp(), player.hp + heal);
        report = `ใช้ ${skill.name} โจมตี ${dmg} และดูด HP กลับ +${heal}`;
      } else if (key === '6') {
        SoundFX.play('stun');
        const dmg = Math.round(totalAtk() * 1.7);
        e.hp -= dmg;
        combat.stunned = true;
        report = `ใช้ ${skill.name} โจมตี ${dmg} และทำให้ศัตรูมึนงง`;
      } else if (key === '7') {
        SoundFX.play('thunder');
        const dmg = Math.round(totalAtk() * 3.0);
        e.hp -= dmg;
        report = `ใช้ ${skill.name} ฟาดสายฟ้าเสียหาย ${dmg}`;
      } else if (key === '8') {
        SoundFX.play('shield');
        combat.guarding = 0.35;
        const heal = Math.round(maxHp() * 0.45);
        player.hp = Math.min(maxHp(), player.hp + heal);
        report = `ใช้ ${skill.name} ฟื้น HP +${heal} และลดดาเมจ 65%`;
      } else if (key === '9') {
        SoundFX.play('ultimate');
        const dmg = Math.round(totalAtk() * 4.2 + maxSp());
        e.hp -= dmg;
        report = `ใช้ ${skill.name} พิฆาตเสียหาย ${dmg}`;
      } else if (key === '0') {
        SoundFX.play('ultimate');
        const dmg = Math.round(totalAtk() * 6.0 + (player.foundationLevel * 25));
        e.hp -= dmg;
        report = `ใช้สุดยอดวิชา ${skill.name} ทำลายล้าง ${dmg}`;
      }

      if (e.hp <= 0) {
        addLog(report, 'c-hit');
        winCombat(report);
        return;
      }

      let enemyReport = '';
      if (combat.stunned) {
        combat.stunned = false;
        enemyReport = `ศัตรูติดมึนงง ข้ามการโจมตี`;
      } else {
        let eDmg = Math.round(e.atk * (0.88 + Math.random() * 0.24));
        if (combat.guarding) eDmg = Math.max(1, Math.round(eDmg * combat.guarding));
        player.hp -= eDmg;
        setTimeout(() => SoundFX.play('hurt'), 220);
        enemyReport = `${e.name} สวนกลับเสีย ${eDmg} HP`;
      }

      if (player.hp <= 0) {
        SoundFX.play('defeat');
        player.hp = Math.round(maxHp() * 0.6);
        player.spirit = Math.round(maxSp() * 0.6);
        const returnCity = player.mapId === 'zone_4' ? 'city_3' : player.mapId === 'zone_3' ? 'city_2' : 'city_1';
        player.mapId = returnCity;
        player.x = 3;
        player.y = 3;
        combat = null;
        SoundFX.updateAmbienceTone();
        saveData();
        renderAll();
        focusGameScreen();
        addLog(`💀 พ่ายแพ้ต่อ ${e.name}! ถูกส่งกลับ ${MAPS[returnCity].name} (ฟื้น HP 60%)`, 'c-bad', true);
        return;
      }

      const fullSpeech = `${report} ศัตรูเหลือ HP ${e.hp} | ${enemyReport} คุณเหลือ HP ${player.hp} จาก ${maxHp()}, จิต ${player.spirit}`;
      addLog(`${report} | ${enemyReport}`, 'c-hit', false);
      announce(fullSpeech);
      saveData();
      renderAll();
    }

    function winCombat(lastHitReport = '') {
      const e = combat.enemy;
      if (combat.spotKey && activeMonsters[player.mapId]) {
        delete activeMonsters[player.mapId][combat.spotKey];
      }
      player.gold += e.gold;
      const drops = [];
      (e.drops || []).forEach(([id, ch]) => {
        if (Math.random() <= ch) {
          addItem(id, 1);
          drops.push(ITEMS[id].name);
        }
      });

      const dropMsg = drops.length ? ` ได้ไอเทม: ${drops.join(', ')}` : '';
      combat = null;
      SoundFX.updateAmbienceTone();
      const prevLv = player.level;
      const lvMsg = gainExp(e.exp);
      if (player.level === prevLv) {

    }

    function scanMap() {
      SoundFX.play('poi');
      const m = MAPS[player.mapId];
      const landmarks = [];
      for (const [key, t] of Object.entries(m.tiles)) {
        const [tx, ty] = key.split(',').map(Number);
        landmarks.push(`${t.label} ที่ (${tx},${ty}) ${directionHint(player.x, player.y, tx, ty)}`);
      }
      if (m.type === 'wild' && activeMonsters[player.mapId]) {
        for (const [key, idx] of Object.entries(activeMonsters[player.mapId])) {
          const [mx, my] = key.split(',').map(Number);
          const mon = MONSTER_POOLS[player.mapId][idx];
          landmarks.push(`อสูร ${mon.name} ที่ (${mx},${my}) ${directionHint(player.x, player.y, mx, my)}`);
        }
      }
      const html = `<strong>${m.name}</strong> พิกัดคุณ (${player.x}, ${player.y})<br>• ` + landmarks.join('<br>• ');
      showInspect('🧭 สแกนรอบตัว [L]', html);
    }

    function directionHint(px, py, tx, ty) {
      const dx = tx - px;
      const dy = ty - py;
      if (dx === 0 && dy === 0) return '(คุณยืนอยู่ตรงนี้ กด Enter เข้าได้เลย)';
      const parts = [];
      if (dy < 0) parts.push(`ขึ้น ${Math.abs(dy)} ก้าว`);
      if (dy > 0) parts.push(`ลง ${dy} ก้าว`);
      if (dx < 0) parts.push(`ซ้าย ${Math.abs(dx)} ก้าว`);
      if (dx > 0) parts.push(`ขวา ${dx} ก้าว`);
      return `(${parts.join(', ')})`;
    }

    function openShopModal(tileInfo, silent = false) {
      let html = `<p style="color:var(--gold); margin-bottom:8px;">🪙 ศิลาของคุณ: <strong>${player.gold}</strong></p>`;
      if (tileInfo.kind === 'shop_sell') {

        if (tileInfo.items?.length) {
          html += `<h4 style="margin-top:10px; color:var(--jade);">สินค้าพิเศษที่ร้านนี้</h4><div class="item-list">` +
            tileInfo.items.map(id => renderBuyCard(ITEMS[id])).join('') + `</div>`;
        }
        const owned = Object.entries(player.inventory).filter(([, q]) => q > 0);
        html += `<h4 style="margin-top:12px; color:var(--gold);">ไอเทมในกระเป๋าของคุณ (กด Enter เพื่อขาย)</h4>`;
        html += owned.length ? `<div class="item-list">` + owned.map(([id, q]) => {
          const it = ITEMS[id];
          return `<div class="item-card"><div><strong>${it.name}</strong> (มี ${q})<br><small style="color:var(--muted)">${it.desc}</small></div>
            <button class="btn btn-sm btn-gold" aria-label="ขาย ${it.name} มี ${q} ชิ้น ได้รับ ${it.sell} ศิลา กด Enter เพื่อขาย" onclick="sellFromShop('${id}')">ขาย 1 ชิ้น (+${it.sell} ศิลา)</button></div>`;
        }).join('') + `</div>` : `<p style="color:var(--muted); margin-top:6px;">กระเป๋าว่างเปล่า</p>`;
      } else {
        html += `<p style="font-size:13px; color:var(--muted);">${tileInfo.desc}</p><div class="item-list">` +
          tileInfo.items.map(id => renderBuyCard(ITEMS[id])).join('') + `</div>`;
      }
      openModal(`🏪 ${tileInfo.label}`, html, `เข้าสู่ ${tileInfo.label} คุณมี ${player.gold} ศิลา กด Tab เลือกสินค้า กด Enter เพื่อซื้อ กด Esc ออก`, silent);
    }

    function renderBuyCard(it) {
      const owned = player.inventory[it.id] || 0;
      const can = player.gold >= it.price;
      return `<div class="item-card">
        <div><strong>${it.name}</strong> (มี ${owned})<br><small style="color:var(--muted)">${it.desc}</small></div>
        <button class="btn btn-sm btn-gold" ${can ? '' : 'disabled'} aria-label="ซื้อ ${it.name} ราคา ${it.price} ศิลา ${it.desc} คุณมีอยู่ ${owned} ชิ้น กด Enter เพื่อซื้อ" onclick="buyFromShop('${it.id}')">ซื้อ (${it.price} ศิลา)</button>
      </div>`;
    }

    function buyFromShop(id) {
      const it = ITEMS[id];
      if (!it || player.gold < it.price) { SoundFX.play('bump'); return; }
      player.gold -= it.price;
      addItem(id, 1);
      SoundFX.play('coin');
      saveData();
      renderHeader();
      addLog(`ซื้อ ${it.name} ราคา ${it.price} ศิลา (เหลือ ${player.gold} ศิลา)`, 'c-win');
      const info = getTileInfo(player.mapId, player.x, player.y);
      openShopModal(info, true);
      announce(`ซื้อ ${it.name} สำเร็จ เหลือ ${player.gold} ศิลา`);
    }

    function sellFromShop(id) {
      const it = ITEMS[id];
      if (!it) return;
      if (player.equippedWeapon === id && player.inventory[id] <= 1) player.equippedWeapon = null;
      if (!removeItem(id, 1)) return;
      player.gold += it.sell;
      SoundFX.play('coin');
      saveData();
      renderHeader();
      addLog(`ขาย ${it.name} ได้รับ +${it.sell} ศิลา`, 'c-win');
      const info = getTileInfo(player.mapId, player.x, player.y);
      openShopModal(info, true);
      announce(`ขาย ${it.name} ได้ +${it.sell} ศิลา รวมมี ${player.gold} ศิลา`);
    }

    function openBagModal(silent = false) {
      const entries = Object.entries(player.inventory).filter(([, q]) => q > 0);
      let html = entries.length ? `<div class="item-list">` + entries.map(([id, q]) => {
        const it = ITEMS[id];
        const eq = player.equippedWeapon === id;
        let btn = '';
        if (it.type === 'pill') btn = `<button class="btn btn-sm btn-jade" aria-label="ใช้ยา ${it.name} มี ${q} ชิ้น ${it.desc} กด Enter" onclick="useBagItem('${id}')">ใช้ยา</button>`;
        if (it.type === 'weapon') btn = eq
          ? `<button class="btn btn-sm" aria-label="ถอดอาวุธ ${it.name} ที่สวมใส่อยู่ กด Enter" onclick="unequipWeapon()">ถอดออก</button>`
          : `<button class="btn btn-sm btn-spirit" aria-label="สวมใส่อาวุธ ${it.name} ${it.desc} กด Enter" onclick="equipWeapon('${id}')">สวมใส่</button>`;

        return `<div class="item-card">
          <div><strong>${it.name}</strong> ${eq ? '(สวมใส่อยู่)' : ''} x${q}<br><small style="color:var(--muted)">${it.desc}</small></div>
          <div style="display:flex; gap:6px;">${btn}</div>
        </div>`;
      }).join('') + `</div>` : `<p style="color:var(--muted)">ยังไม่มีไอเทมในกระเป๋า</p>`;

    function useBagItem(id) {

      player.spirit = Math.min(maxSp(), player.spirit + it.boost.sp);

      document.getElementById('uiAtk').textContent = totalAtk();

      const mHp = maxHp(), mSp = maxSp();
      document.getElementById('uiHp').textContent = `${player.hp}/${mHp}`;
      document.getElementById('barHp').style.width = `${Math.min(100, (player.hp / mHp) * 100)}%`;
      document.getElementById('uiSp').textContent = `${player.spirit}/${mSp}`;
      document.getElementById('barSp').style.width = `${Math.min(100, (player.spirit / mSp) * 100)}%`;
      document.getElementById('uiExp').textContent = `${player.exp}/${player.maxExp}`;
      document.getElementById('barExp').style.width = `${Math.min(100, (player.exp / player.maxExp) * 100)}%`;
    }

    function renderMap() {
      const m = MAPS[player.mapId];
      document.getElementById('exploreView').classList.toggle('hidden', !!combat);
      document.getElementById('combatView').classList.toggle('hidden', !combat);
      if (combat) return;

      document.getElementById('uiMapTitle').textContent = `🗺️ ${m.name} (${m.size}×${m.size})`;
      document.getElementById('uiCoords').textContent = `พิกัด (${player.x}, ${player.y})`;
      document.getElementById('btnSearchMonster').classList.toggle('hidden', m.type !== 'wild');

      const curInfo = getTileInfo(player.mapId, player.x, player.y);
      document.getElementById('uiTileDesc').textContent = `${curInfo.label}: ${curInfo.desc}`;

      const grid = document.getElementById('squareGrid');
      let html = '';
      for (let y = 0; y < m.size; y++) {
        for (let x = 0; x < m.size; x++) {
          const isPlayer = (x === player.x && y === player.y);
          const info = getTileInfo(player.mapId, x, y);
          let cls = m.type === 'city' ? 'city-ground' : 'wild-ground';
          if (info.kind.startsWith('shop_')) cls = 'shop-tile';
          if (info.kind === 'gate') cls = 'gate-tile';
          if (info.kind === 'inn') cls = 'inn-tile';
          if (info.kind === 'foundation') cls = 'foundation-tile';
          if (info.kind === 'monster') cls = 'monster-tile';
          if (isPlayer) cls += ' player-here';

          const icon = isPlayer ? '🧙' : info.icon;
          const shortLbl = isPlayer ? 'คุณ' : (info.kind === 'empty' ? '' : info.label.slice(0, 6));
          html += `<div class="tile ${cls}" onclick="clickTile(${x},${y})">
            <span>${icon}</span>
            ${shortLbl ? `<span class="tile-sub">${shortLbl}</span>` : ''}
          </div>`;
        }
      }
      grid.innerHTML = html;
    }

    function clickTile(tx, ty) {
      if (combat || modalOpen) return;
      if (tx === player.x && ty === player.y) {
        interactCurrentTile();
        return;
      }
      const dx = Math.sign(tx - player.x);
      const dy = Math.sign(ty - player.y);
      if (Math.abs(tx - player.x) >= Math.abs(ty - player.y)) movePlayer(dx, 0);
      else movePlayer(0, dy);
    }

    function renderCombat() {
      if (!combat) return;
      const e = combat.enemy;
      document.getElementById('cEnemyEmoji').textContent = e.emoji;
      document.getElementById('cEnemyName').textContent = e.name;
      document.getElementById('cEnemyMeta').textContent = `Lv.${e.lv} | พลังโจมตี: ${e.atk}${combat.stunned ? ' | 💫 ติดมึนงง' : ''}`;
      document.getElementById('cEnemyHp').textContent = `${Math.max(0, e.hp)}/${e.maxHp}`;
      document.getElementById('cEnemyHpBar').style.width = `${Math.max(0, (e.hp / e.maxHp) * 100)}%`;

      document.getElementById('combatSkillGrid').innerHTML = SKILLS.map(s => {
        const unlocked = player.level >= s.unlockLv;
        const enoughSp = player.spirit >= s.cost;
        const canUse = unlocked && enoughSp;
        const aria = unlocked
          ? `กดเลข ${s.key} ${s.name} ใช้จิต ${s.cost} ${s.desc}`
          : `กดเลข ${s.key} ${s.name} ล็อกอยู่ ต้องการเลเวล ${s.unlockLv}`;
        return `<button class="skill-btn" ${canUse ? '' : 'disabled'} aria-label="${aria}" onclick="useSkillByKey('${s.key}')">
          <div><span class="skill-key">${s.key}</span><strong>${s.name}</strong></div>
          <div style="font-size:11px; color:var(--muted);">
            ${unlocked ? `ใช้จิต: ${s.cost} | ${s.desc}` : `🔒 ปลดล็อกที่ Lv.${s.unlockLv}`}
          </div>
        </button>`;
      }).join('');
    }

    // ==================== GLOBAL KEYBOARD SHORTCUTS (ENTER ONLY, NO SPACEBAR, NO V/B) ====================
    document.addEventListener('keydown', (e) => {
      if (!player) return;
      const tag = e.target.tagName;
      if (tag === 'INPUT' || tag === 'TEXTAREA' || tag === 'SELECT') return;

      if (e.key === ' ' || e.code === 'Space') {
        e.preventDefault();
        return;
      }

      // Volume Controls work everywhere (map, combat, and inside modals!)
      if (e.key === 'PageUp')   { e.preventDefault(); adjustBgmVolume(0.1); return; }
      if (e.key === 'PageDown') { e.preventDefault(); adjustBgmVolume(-0.1); return; }
      if (e.key === 'Home')     { e.preventDefault(); adjustMasterVolume(0.1); return; }
      if (e.key === 'End')      { e.preventDefault(); adjustMasterVolume(-0.1); return; }

      if (e.key === 'Escape') {
        if (modalOpen) { e.preventDefault(); closeModal(); }
        return;
      }

      if (modalOpen) return;

      const k = e.key.toLowerCase();

      if (combat) {
        if (/^[0-9]$/.test(e.key)) {
          e.preventDefault();
          useSkillByKey(e.key);
          return;
        }
        if (k === 'r') { e.preventDefault(); fleeCombat(); return; }
      }

      if (!combat) {
        if (e.key === 'ArrowUp')    { e.preventDefault(); movePlayer(0, -1); return; }
        if (e.key === 'ArrowDown')  { e.preventDefault(); movePlayer(0, 1); return; }
        if (e.key === 'ArrowLeft')  { e.preventDefault(); movePlayer(-1, 0); return; }
        if (e.key === 'ArrowRight') { e.preventDefault(); movePlayer(1, 0); return; }
        if (e.key === 'Enter' && tag !== 'BUTTON') {
          e.preventDefault();
          interactCurrentTile();
          return;
        }
        if (k === 'a') { e.preventDefault(); searchNearbyMonster(); return; }
      }

      if (k === 'c') { e.preventDefault(); inspectStatus(); }
      else if (k === 'l') { e.preventDefault(); scanMap(); }
      else if (k === 'm') { e.preventDefault(); inspectMapInfo(); }
      else if (k === 'i') { e.preventDefault(); openBagModal(); }
      else if (k === 'f') { e.preventDefault(); openFoundationModal(); }
      else if (k === 'k') { e.preventDefault(); openSkillsModal(); }
      else if (k === 'j') { e.preventDefault(); readRecentLog(); }
      else if (k === 'q') { e.preventDefault(); quickHealHp(); }
      else if (k === 'w') { e.preventDefault(); quickHealSp(); }
      else if (k === 's') { e.preventDefault(); manualSave(); }
      else if (k === 'h' || e.key === '?') { e.preventDefault(); openHelpModal(); }
    });

    document.addEventListener('keyup', (e) => {
      if (!player) return;
      const tag = e.target.tagName;
      if (tag === 'INPUT' || tag === 'TEXTAREA' || tag === 'SELECT') return;
      if (e.key === ' ' || e.code === 'Space') {
        e.preventDefault();
      }
    });

      const saved = localStorage.getItem(STORAGE_CURRENT);
      if (saved) {
        const acc = getAccounts();
        if (acc[saved]?.data) startSession(saved, acc[saved].data);
      }
    });
